   1: <?php
   2: 
   3: namespace App\Services;
   4: 
   5: use App\Models\Movement;
   6: use App\Models\Product;
   7: use App\Models\User;
   8: use Illuminate\Support\Facades\DB;
   9: use Illuminate\Validation\ValidationException;
  10: 
  11: class InventoryService
  12: {
  13:     public function createProduct(array $data, User $user): Product
  14:     {
  15:         $initialStock = (int) $data['initial_stock'];
  16:         if ($initialStock < 0 || $initialStock > Product::MAX_STOCK) {
  17:             throw ValidationException::withMessages(['initial_stock' => 'El stock inicial está fuera del rango permitido.']);
  18:         }
  19: 
  20:         return DB::transaction(function () use ($data, $initialStock, $user) {
  21:             unset($data['initial_stock'], $data['stock'], $data['is_active']);
  22:             $product = Product::create($data);
  23:             if ($initialStock > 0) {
  24:                 $this->record($product, Movement::ENTRY, $initialStock, $user, 'Stock inicial');
  25:             }
  26: 
  27:             return $product->refresh();
  28:         }, attempts: 5);
  29:     }
  30: 
  31:     public function record(Product $product, string $type, int $quantity, User $user, ?string $note = null): Movement
  32:     {
  33:         if (! in_array($type, [Movement::ENTRY, Movement::EXIT], true)) {
  34:             throw ValidationException::withMessages(['type' => 'Selecciona una entrada o salida.']);
  35:         }
  36:         if ($quantity <= 0 || $quantity > Product::MAX_STOCK) {
  37:             throw ValidationException::withMessages(['quantity' => 'La cantidad debe ser un entero positivo dentro del rango permitido.']);
  38:         }
  39: 
  40:         return DB::transaction(function () use ($product, $type, $quantity, $user, $note) {
  41:             // PHP 8.3 uses DEFERRED transactions even with Laravel's IMMEDIATE setting.
  42:             // Make the first statement a write: SQLite locks writers before we read the balance.
  43:             DB::table('products')->where('id', $product->id)->update(['stock' => DB::raw('stock')]);
  44:             $current = Product::findOrFail($product->id);
  45:             if (! $current->is_active) {
  46:                 throw ValidationException::withMessages(['product_id' => 'El producto está inactivo. Selecciona un producto activo.']);
  47:             }
  48:             $previous = $current->stock;
  49:             $next = $previous + ($type === Movement::ENTRY ? $quantity : -$quantity);
  50:             if ($next < 0) {
  51:                 throw ValidationException::withMessages(['quantity' => "Stock insuficiente. Solo hay {$previous} unidades disponibles."]);
  52:             }
  53:             if ($next > Product::MAX_STOCK) {
  54:                 throw ValidationException::withMessages(['quantity' => 'La entrada supera el límite de unidades permitido.']);
  55:             }
  56:             // A conditional write also prevents overwriting a newer balance with a stale read.
  57:             $updated = Product::whereKey($current->id)->where('is_active', true)->where('stock', $previous)
  58:                 ->update(['stock' => $next, 'updated_at' => now()]);
  59:             if ($updated !== 1) {
  60:                 throw ValidationException::withMessages(['quantity' => 'El inventario cambió. Actualiza la página e intenta nuevamente.']);
  61:             }
  62: 
  63:             return Movement::create([
  64:                 'product_id' => $current->id, 'user_id' => $user->id, 'type' => $type,
  65:                 'quantity' => $quantity, 'previous_stock' => $previous, 'resulting_stock' => $next,
  66:                 'note' => $note, 'created_at' => now(),
  67:             ]);
  68:         }, attempts: 5);
  69:     }
  70: }
